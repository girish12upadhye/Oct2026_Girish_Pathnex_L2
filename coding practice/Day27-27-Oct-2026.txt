# Day 27 — Serverless Scaling with Lambda and DynamoDB

# Ansible - Deploy Serverless Lambda Functions
- name: Deploy Serverless Lambda Function
  hosts:
  tasks:
    - name: Create Lambda function
      command: aws lambda create-function --function-name girishtechverse-lambda --runtime nodejs14.x --role arn:aws:iam:123456789012:role/lambda-role --handler index.handler --zip-file fileb://lambda.zip

# Terraform - Create DynamoDB Table for Serverless Application
resource "aws_dynamodb_table" "girishtechverse_table" {
  name		 = "girishtechverse-table"
  hash-key	 = "id"
  billing_mode   = "PAY_PER_REQUEST"
  attribute {
    name = "id"
    type = "S"
  }
}    

# Kubernetes - Deploy Serverless Using Kubeless
apiVersion: kubeless.io/v1beta1
kind: Function
metadata:
  name: girishtechverse-lambda
  namespace: default
spec:
  runtime: nodejs14
  handler: index.handler
  function: |
    module.exports = function(event, context){
      context.succeed('Hello World');
};

# Jenkinsfile - Lambda Function Deployment Pipeline
pipeline {
    agent any
    stages {
	stage('Build') {
	    steps {
		echo 'Building Lambda function package....'
		sh 'zip -r lambda.zip lambda/'
	    }
	}
	stage('Deploy to Lambda'){
	    steps {
		script {
		    sh 'aws lambda update-function-code --function-name girishtechverse-lambda --zip-file fileb://lambda.zip'
		}
	    }
	}
    }
}
