# Terraform - Provisioning Multi-Tier Application with VPC and Security
resource "aws_vpc" "girishtechverse_vpc" {
  cidr_block = "10.0.0.0/16"
}

resource "aws_instance" "girishtechverse_app" {
  ami		= "ami-0abcd1234abcd1234"
  instance_type = "t2.medium"
  subnet_id	= aws_subnet.girishtechverse_subnet.id
  security_groups = [aws_security_group.girishtechverse_sg.name]
}

# Kubernetes - Deploy Full Application Stack
apiVersion: apps/v1
kind: Deployment
metadata:
  name: girishtechverse-app
spec:
  replicas: 2
  selector:
    matchLabels:
      app: girishtechverse-app
  template:
    metadata:
      labels:
	app: girishtechverse-app
    spec:
      containers:
	- name: nginx
	  image: nginx
	  ports:
	    - containersPort: 80
---

apiVersion: v1
kind: Service
metadata:
  name: girishtechverse-service
spec:
  selector:
    app: girishtechverse-app
  ports:
    - port: 80
      targetPort: 80
