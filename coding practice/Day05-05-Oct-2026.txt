# Ansible - Create User and Set Permission
- name: Create a new user
  hosts: all
  become: yes

  tasks:
    - name: Create user
      user:
        name: girishtechverse
        state: present
        shell: /bin/bash
    - name: Set Permissions for the user
      file:
        path: /home/girishtechverse
        owner: girishtechverse
        group: girishtechverse
        mode: '0755'
