# Ansible - Multi-Node Nginx Setup
- name: Setup Nginx on multiple nodes
  hosts: all
  become: yes
  tasks:
    - name: Install nginx
      yum:
	name: nginx
	state: present

    - name: Start Nginx Service
      service:
	name: nginx
	state: started
	enabled: yes

        index index.html index.html;

  name: girish-nginx

      app: girishtechverse-nginx

        app: girishtechverse-nginx

      containers:

	  image: nginx
	  volumeMounts:
	    - mountPath: /etc/nginx/nginx.conf
	      name: nginx-conf
	      subPath: nginx.conf
      volumes:
        - name: nginx-config
	  configMap:
	    name: nginx-config
	- name: nginx-secret
	  secret:
	    secretName: nginx-secret

# Jenkinsfile - Multi-Branch Pipeline with Docker Build
pipeline {
    agent any
    stages {
	stage('Build') {
	    steps {
		script {
		    docker.build('girishtechverse-app')
		}
	    }
	}
	stage('Deploy') {
	    steps {
		script {
		    kubectl apply -f kubernetes/deployment.yaml
		}
	    }
	}
    }
}

# GitLab CI/CD - Advanced Deployment with Helm
stages:
  - build
  - push
  - deploy

build:
  stage: build
  script:
    - docker build -t girishtechverse-web-app .

push:
  stage: push
  script:
    - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
    - docker push girishtechverse-web-app

deploy:
  stage: deploy
  script:
    - helm repo add pathnex https://charts.girishtechverse.com
    - helm install girishtechverse-nginx girishtechverse/nginx-ingress
